The background is a search

Each click runs a breadth-first search from the node nearest your cursor. Distance from the start decides when a node lights up, so the wave you see is the queue emptying one level at a time.

When the page loads it starts from three nodes at once, which is the multi-source version. The page does this in JavaScript; here it is in C++.

vector<int> bfs(vector<vector<int>>& g, vector<int>& src) {
    vector<int> d(g.size(), -1);
    queue<int> q;
    for (int s : src) {
        d[s] = 0;
        q.push(s);
    }
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : g[u]) {
            if (d[v] != -1) continue;
            d[v] = d[u] + 1;
            q.push(v);
        }
    }
    return d;
}
